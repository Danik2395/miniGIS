25. для точки пересечения отрезков алгоритм примерно такой
    (не делать проверку через вызов is_intercest, потому что тогда получится,
    что эта общая логика проверок вызывается два раза, а она и так для счёта нужна):

p = вектор первого отрезка
q = вектор второго отрезка
cross_product(p, q) чтобы разделить счёт на два случая: отрезки коллинеарны, отрезки не поллинеарны

if отрезки не коллинеарны (cross_product != 0):
    из системы уравнений найти коэффициенты для обоих уравнений прямой (от начал отрезков)

    if они одновременно 0 <= coeff <= 1:
        точка лежит в отрезках -> пересечение есть.
    else:
        return нулевую геометрию

    подставить один из коэффициентов в соответствующее уравнение прямой
    return точку пересечения

else if отрезки коллинеарны (cross_product == 0):
    d = вектор от начала 1 отрезка до начала 2
    r = отрезок от начала 1 до конца 2

    cross_product(p, d)

    if cross_product != 0:
        отрезки параллелельны, но лежат не на одной прямой -> пересечения нет
        return нулевую геометрию

    t1 = коэффициент проекции d на p
    t2 = коэффициент проекции r на p

    если отрезки внахлёст, то отрезок пересечения лежит только на первом отрезке
    start = max(0, min(t1, t2))
    end   = min(1, max(t1, t2))

    if start > end:
        отрезки на одной прямой, но пересечений нет
        return нулевую геометрию

    if start == end:
        отрзеки соприкасаются в одной точке
        return (точка начала 1 отрезка) + p * start

    отрезки внахлёст -> нужно вернуть сегмент

    p1 = (точка начала 1 отрезка) + p * start
    p2 = (точка начала 1 отрезка) + p * end

    return сегмент(p1, p2)
